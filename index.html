function automaticFeedWater(time){ const needed = game.chickens * CONFIG.feedPerChickenPerMeal; let feedOK = false; /* بررسی خوراک */ if(game.feed >= needed){ game.feed -= needed; feedOK = true; game.feedShortageDays = 0; }else{ game.feedShortageDays++; } /* بررسی آب آب در هر وعده باید تأمین شود. */ if(game.water > 0){ game.water = 100; game.waterShortageDays = 0; }else{ game.waterShortageDays++; } /* اگر هر دو مورد درست باشند، تولید تخم مرغ انجام می‌شود. */ if(feedOK && game.waterShortageDays === 0){ const produced = produceEggs(); showToast( "⏰ وعده " + time + " انجام شد؛ " + formatNumber(produced) + " تخم‌مرغ تولید شد" ); }else{ showToast( "⚠️ وعده " + time + " با مشکل خوراک یا آب مواجه شد" ); applyShortageEffects(); } saveSilently(); render(); }
